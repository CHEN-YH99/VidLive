#### 前端实现（Next.js）
```typescript
// app/upload/page.tsx
'use client';

import { useState } from 'react';
import { useDropzone } from 'react-dropzone';

export default function UploadPage() {
  const [uploadProgress, setUploadProgress] = useState(0);
  const [videoUrl, setVideoUrl] = useState<string | null>(null);

  const { getRootProps, getInputProps } = useDropzone({
    accept: { 'video/*': ['.mp4', '.mov', '.avi', '.mkv'] },
    maxSize: 500 * 1024 * 1024, // 500MB
    onDrop: async (files) => {
      const file = files[0];
      await uploadVideo(file);
    },
  });

  const uploadVideo = async (file: File) => {
    const formData = new FormData();
    formData.append('video', file);

    const xhr = new XMLHttpRequest();
    xhr.upload.addEventListener('progress', (e) => {
      setUploadProgress(Math.round((e.loaded / e.total) * 100));
    });

    xhr.open('POST', '/api/upload');
    xhr.send(formData);

    xhr.onload = () => {
      const result = JSON.parse(xhr.responseText);
      setVideoUrl(result.videoUrl);
    };
  };

  return (
    <div {...getRootProps()} className="border-2 border-dashed p-8">
      <input {...getInputProps()} />
      <p>拖拽视频到这里，或点击选择文件</p>
      {uploadProgress > 0 && <p>上传进度: {uploadProgress}%</p>}
    </div>
  );
}
```

```typescript
      status: 'pending',
    });

      conversionId: conversion.id,
      videoUrl: `/uploads/${filename}`,
      duration: metadata.duration,
    };
  });
```

### 2. 智能关键帧提取

#### 算法实现（MVP版本）
```typescript
// src/services/keyframe-detector.ts
import { exec } from 'child_process';
import { promisify } from 'util';
import cv from '@u4/opencv4nodejs'; // OpenCV Node绑定

const execAsync = promisify(exec);

export class KeyframeDetector {
  /**
   * 提取视频的所有I帧（关键帧）
   */
  async extractIFrames(videoPath: string, outputDir: string): Promise<string[]> {
    const cmd = `ffmpeg -i "${videoPath}" -vf "select='eq(pict_type,I)'" -vsync 0 "${outputDir}/frame_%04d.jpg"`;
    await execAsync(cmd);

    // 返回生成的图片路径列表
    const files = await fs.readdir(outputDir);
    return files.map(f => path.join(outputDir, f));
  }

  /**
   * 计算图片清晰度（Laplacian方差）
   */
  calculateSharpness(imagePath: string): number {
    const img = cv.imread(imagePath);
    const gray = img.cvtColor(cv.COLOR_BGR2GRAY);
    const laplacian = gray.laplacian(cv.CV_64F);
    const { stddev } = laplacian.meanStdDev();
    return stddev.w * stddev.w; // 方差越大越清晰
  }

  /**
   * 人脸检测（可选增强）
   */
  async detectFaces(imagePath: string): Promise<number> {
    const img = cv.imread(imagePath);
    const classifier = new cv.CascadeClassifier(cv.HAAR_FRONTALFACE_ALT2);
    const { objects } = classifier.detectMultiScale(img.bgrToGray());
    return objects.length;
  }

  /**
   * 智能推荐最佳关键帧
   */
  async recommendKeyframes(videoPath: string, topN: number = 3): Promise<RecommendedFrame[]> {
    const tempDir = `/tmp/${uuidv4()}`;
    await fs.mkdir(tempDir, { recursive: true });

    // 1. 提取所有I帧
    const frames = await this.extractIFrames(videoPath, tempDir);

    // 2. 评分计算
    const scored = await Promise.all(
      frames.map(async (framePath, idx) => {
        const sharpness = this.calculateSharpness(framePath);
        const faceCount = await this.detectFaces(framePath);

        // 综合评分：清晰度 + 人脸加权
        const score = sharpness + faceCount * 1000;

        return {
          framePath,
          frameIndex: idx,
          timestamp: this.getTimestamp(videoPath, idx),
          score,
          sharpness,
          faceCount,
        };
      })
    );

    // 3. 排序并返回Top N
    return scored.sort((a, b) => b.score - a.score).slice(0, topN);
  }

  /**
   * 根据帧索引计算时间戳
   */
  private async getTimestamp(videoPath: string, frameIndex: number): Promise<number> {
    const { stdout } = await execAsync(
      `ffprobe -v error -select_streams v:0 -show_entries stream=r_frame_rate -of csv=p=0 "${videoPath}"`
    );
    const [num, den] = stdout.trim().split('/').map(Number);
    const fps = num / den;
    return frameIndex / fps;
  }
}
```

### 3. Live Photo 生成核心

#### 格式规范说明
Live Photo由两个文件组成：
- **静态图片**：HEIC格式（或JPEG兼容）
- **视频片段**：MOV格式，H.264编码，AAC音频

**关键元数据（必须）**：
```
com.apple.quicktime.content.identifier: 唯一UUID
com.apple.quicktime.live-photo.auto: 1
com.apple.quicktime.make: Apple
```

#### 生成器实现
```typescript
// src/services/livephoto-generator.ts
import { exec } from 'child_process';
import { promisify } from 'util';
import { v4 as uuidv4 } from 'uuid';
import archiver from 'archiver';

const execAsync = promisify(exec);

export class LivePhotoGenerator {
  /**
   * 生成Live Photo
   * @param videoPath 原始视频路径
   * @param keyframeTime 关键帧时间点（秒）
   * @param duration Live Photo视频时长（秒，默认3秒）
   */
  async generate(
    videoPath: string,
    keyframeTime: number,
    duration: number = 3
  ): Promise<{ photoPath: string; videoPath: string; zipPath: string }> {
    const workDir = `/tmp/livephoto_${uuidv4()}`;
    await fs.mkdir(workDir, { recursive: true });

    // 1. 提取关键帧为静态图片
    const photoPath = path.join(workDir, 'photo.jpg');
    await execAsync(
      `ffmpeg -i "${videoPath}" -ss ${keyframeTime} -frames:v 1 -q:v 2 "${photoPath}"`
    );

    // 2. 转换为HEIC格式（可选，需要libheif支持）
    const heicPath = path.join(workDir, 'photo.heic');
    try {
      await execAsync(`heif-enc -q 90 "${photoPath}" -o "${heicPath}"`);
    } catch {
      // 降级：如果HEIC转换失败，使用JPEG
      console.warn('HEIC conversion failed, fallback to JPEG');
      await fs.copyFile(photoPath, heicPath.replace('.heic', '.jpg'));
    }

    // 3. 裁剪视频片段（关键帧前后各1.5秒）
    const videoStart = Math.max(0, keyframeTime - 1.5);
    const movPath = path.join(workDir, 'video.mov');
    await execAsync(
      `ffmpeg -i "${videoPath}" -ss ${videoStart} -t ${duration} \
       -c:v libx264 -preset fast -crf 23 \
       -c:a aac -b:a 128k -ar 44100 \
       -movflags +faststart \
       "${movPath}"`
    );

    // 4. 注入Live Photo元数据（关键步骤！）
    const contentId = uuidv4().toUpperCase();
    await this.injectMetadata(movPath, heicPath, contentId);

    // 5. 打包为ZIP
    const zipPath = path.join(workDir, 'livephoto.zip');
    await this.createZip(heicPath, movPath, zipPath);

    return {
      photoPath: heicPath,
      videoPath: movPath,
      zipPath,
    };
  }

  /**
   * 注入Live Photo元数据
   */
  private async injectMetadata(
    movPath: string,
    heicPath: string,
    contentId: string
  ): Promise<void> {
    // MOV文件元数据
    const movMetadata = [
      `-QuickTime:ContentIdentifier="${contentId}"`,
      `-QuickTime:LivePhotoAuto=1`,
      `-QuickTime:Make=Apple`,
      `-QuickTime:Model="iPhone 14 Pro"`, // 伪装设备
    ].join(' ');

    await execAsync(`exiftool -overwrite_original ${movMetadata} "${movPath}"`);

    // HEIC文件元数据（保持一致）
    const heicMetadata = [
      `-XMP:LivePhotoVideoIndex=0`,
      `-XMP:ContentIdentifier="${contentId}"`,
    ].join(' ');

    await execAsync(`exiftool -overwrite_original ${heicMetadata} "${heicPath}"`);
  }

  /**
   * 打包为ZIP文件
   */
  private async createZip(
    photoPath: string,
    videoPath: string,
    zipPath: string
  ): Promise<void> {
    return new Promise((resolve, reject) => {
      const output = fs.createWriteStream(zipPath);
      const archive = archiver('zip', { zlib: { level: 9 } });

      output.on('close', resolve);
      archive.on('error', reject);

      archive.pipe(output);
      archive.file(photoPath, { name: path.basename(photoPath) });
      archive.file(videoPath, { name: path.basename(videoPath) });
      archive.finalize();
    });
  }
}
```


### 4. 异步任务队列

#### BullMQ队列配置
```typescript
// src/queues/conversion-queue.ts
import { Queue, Worker, Job } from 'bullmq';
import { Redis } from 'ioredis';

const connection = new Redis({
  host: process.env.REDIS_HOST,
  port: parseInt(process.env.REDIS_PORT!),
  maxRetriesPerRequest: null,
});

// 创建任务队列
export const conversionQueue = new Queue('video-conversion', {
  connection,
  defaultJobOptions: {
    attempts: 3, // 失败重试3次
    backoff: { type: 'exponential', delay: 5000 },
    removeOnComplete: { count: 100 }, // 保留最近100条成功记录
    removeOnFail: { count: 500 },
  },
});

// Worker处理器
const worker = new Worker(
  'video-conversion',
  async (job: Job) => {
    const { conversionId, videoPath, keyframeTime, duration } = job.data;

    // 更新状态为处理中
    await db.conversions.update(conversionId, { status: 'processing' });

    try {
      // 生成Live Photo
      const generator = new LivePhotoGenerator();
      const result = await generator.generate(videoPath, keyframeTime, duration);

      // 上传到对象存储
      const downloadUrl = await uploadToR2(result.zipPath);

      // 更新数据库
      await db.conversions.update(conversionId, {
        status: 'completed',
        outputUrl: downloadUrl,
        processingTime: Date.now() - job.timestamp,
        expiresAt: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000), // 7天后过期
      });

      // 清理临时文件
      await fs.rm(path.dirname(result.zipPath), { recursive: true });

      return { success: true, downloadUrl };
    } catch (error) {
      await db.conversions.update(conversionId, {
        status: 'failed',
        errorMessage: error.message,
      });
      throw error;
    }
  },
  { connection, concurrency: 5 } // 最多5个并发任务
);

worker.on('completed', (job) => {
  console.log(`Job ${job.id} completed`);
});

worker.on('failed', (job, err) => {
  console.error(`Job ${job?.id} failed:`, err);
});
```

### 5. 对象存储集成（Cloudflare R2）

#### R2 SDK封装
```typescript
// src/services/storage.ts
import { S3Client, PutObjectCommand, GetObjectCommand } from '@aws-sdk/client-s3';
import { getSignedUrl } from '@aws-sdk/s3-request-presigner';
import fs from 'fs/promises';

export class StorageService {
  private s3Client: S3Client;
  private bucketName: string;

  constructor() {
    this.s3Client = new S3Client({
      region: 'auto',
      endpoint: process.env.R2_ENDPOINT, // https://xxx.r2.cloudflarestorage.com
      credentials: {
        accessKeyId: process.env.R2_ACCESS_KEY_ID!,
        secretAccessKey: process.env.R2_SECRET_ACCESS_KEY!,
      },
    });
    this.bucketName = process.env.R2_BUCKET_NAME!;
  }

  /**
   * 上传文件到R2
   */
  async upload(localPath: string, key: string): Promise<string> {
    const fileContent = await fs.readFile(localPath);

    await this.s3Client.send(
      new PutObjectCommand({
        Bucket: this.bucketName,
        Key: key,
        Body: fileContent,
        ContentType: this.getContentType(localPath),
      })
    );

    // 返回公开访问URL
    return `${process.env.R2_PUBLIC_URL}/${key}`;
  }

  /**
   * 生成临时下载链接（7天有效）
   */
  async getSignedDownloadUrl(key: string): Promise<string> {
    const command = new GetObjectCommand({
      Bucket: this.bucketName,
      Key: key,
    });

    return getSignedUrl(this.s3Client, command, { expiresIn: 7 * 24 * 3600 });
  }

  private getContentType(filePath: string): string {
    const ext = path.extname(filePath).toLowerCase();
    const types: Record<string, string> = {
      '.zip': 'application/zip',
      '.mov': 'video/quicktime',
      '.heic': 'image/heic',
      '.jpg': 'image/jpeg',
    };
    return types[ext] || 'application/octet-stream';
  }
}

// 使用示例
export async function uploadToR2(localPath: string): Promise<string> {
  const storage = new StorageService();
  const key = `livefotos/${uuidv4()}/${path.basename(localPath)}`;
  return storage.upload(localPath, key);
}
```

```typescript
import { useState, useRef, useEffect } from 'react';
```

### 转换进度组件
```typescript
// components/ConversionProgress.tsx
'use client';

import { useEffect, useState } from 'react';

interface ConversionProgressProps {
  conversionId: string;
  onComplete: (downloadUrl: string) => void;
}

export function ConversionProgress({
  conversionId,
  onComplete,
}: ConversionProgressProps) {
  const [status, setStatus] = useState<'processing' | 'completed' | 'failed'>('processing');
  const [progress, setProgress] = useState(0);

  useEffect(() => {
    const interval = setInterval(async () => {
      const res = await fetch(`/api/convert/${conversionId}/status`);
      const data = await res.json();

      setStatus(data.status);

      if (data.status === 'completed') {
        clearInterval(interval);
        onComplete(data.outputUrl);
      } else if (data.status === 'failed') {
        clearInterval(interval);
      } else {
        setProgress((prev) => Math.min(prev + 10, 90));
      }
    }, 1000);

    return () => clearInterval(interval);
  }, [conversionId]);

  return (
    <div className="space-y-4">
      <div className="relative w-full h-4 bg-gray-200 rounded-full overflow-hidden">
        <div
```

## 安全措施

### 1. 文件验证
```typescript
// 检查真实文件类型（防止伪造扩展名）
import fileType from 'file-type';

async function validateVideo(filePath: string): Promise<boolean> {
  const type = await fileType.fromFile(filePath);
  
  const allowedTypes = ['video/mp4', 'video/quicktime', 'video/x-msvideo'];
  
  if (!type || !allowedTypes.includes(type.mime)) {
    throw new Error('Invalid video format');
  }
  
  return true;
}
```

## 技术债务与风险

### 已知限制
1. **Live Photo格式兼容性**
   - 风险：苹果未公开标准，可能因系统更新失效
   - 缓解：持续监控iOS新版本，社区跟进

2. **视频处理性能**
   - 风险：大文件处理慢，用户体验差
   - 缓解：硬件加速、预设质量档位、提示预估时间

3. **存储成本**
   - 风险：用户上传大量视频，R2费用攀升
   - 缓解：严格清理策略、压缩存储、付费用户分摊

### 技术选型风险
- **FFmpeg依赖**：需要系统级安装，Docker部署必须
- **OpenCV性能**：Node.js版性能一般，后续可能需要Python重写
- **Cloudflare R2**：相对新的服务，稳定性待验证
